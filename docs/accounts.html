# Create a self-contained Bitty Box link via REST API
curl -X POST https://bittybox.org/api/bitty/create \
  -H "Content-Type: application/json" \
  -H "X-API-Key: bb_live_YOUR_API_KEY" \
  -d '{
    "title": "System Report",
    "content": "# Live Server Status\nAll systems operational.",
    "format": "markdown"
  }'
const res = await fetch("https://bittybox.org/api/bitty/create", {
  method: "POST",
  headers: {
    "Content-Type": "application/json",
    "X-API-Key": "bb_live_YOUR_API_KEY"
  },
  body: JSON.stringify({
    title: "Interactive Widget",
    content: "<h1>Hello World</h1><script>console.log('running')</script>",
    format: "html"
  })
});
const { url } = await res.json();
console.log("Generated Capsule URL:", url);
import requests

res = requests.post(
    "https://bittybox.org/api/bitty/create",
    headers={"X-API-Key": "bb_live_YOUR_API_KEY"},
    json={
        "title": "Data Analysis",
        "content": "# Report\nSummary data here...",
        "format": "markdown"
    }
)
data = res.json()
print("Capsule Link:", data["url"])
